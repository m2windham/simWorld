// Lane-private diagnostic (storyteller-pressure, baseline task). Replays the storyteller world for a seed and
// follows (a) every wound infection on a citizen from onset to its end and (b) every fire and every burn.
// Reads only: no Rand draws, no jobs, no reservations.
using System;
using System.Collections.Generic;
using System.Globalization;
using System.Linq;

using SimWorld.AI;
using SimWorld.Content;
using SimWorld.Defs;
using SimWorld.Director;
using SimWorld.God.View;
using SimWorld.Health;
using SimWorld.Pawns;
using SimWorld.Scenario;
using SimWorld.Sim;
using SimWorld.Things;

internal sealed class Inf
{
    public Pawn P = null!;
    public Hediff H = null!;
    public int Onset;
    public string JobAtOnset = "";
    public int FirstTended = -1;
    public int Tends;
    public float LastTotalQ;
    public float SumQ;
    public int Samples;
    public int Tended;
    public int InBed;
    public int Asleep;
    public int Downed;
    public float MaxSev;
    public float ImmAtEnd;
    public float SevAtEnd;
    public float GainSum;
    public int End = -1;
    public string Outcome = "running";
    public List<string> Timeline = new List<string>();
}

internal static class Program
{
    private static string F(float v) => v.ToString("0.00", CultureInfo.InvariantCulture);

    private static int Main(string[] args)
    {
        int seed = int.Parse(args[0], CultureInfo.InvariantCulture);
        int days = int.Parse(args[1], CultureInfo.InvariantCulture);
        int band = int.Parse(args[2], CultureInfo.InvariantCulture);
        bool threats = bool.Parse(args[3]);
        string data = args[4];
        string mode = args.Length > 5 ? args[5] : "infection";

        var database = new DefDatabase();
        DefLoadResult result = CoreContent.Load(database, new DefTypeResolver(), new DefLoadOptions { BindDefOfs = true }, data);
        if (!result.Success) throw new InvalidOperationException("content failed to load");
        DefDatabase.Global = database;

        Find.Reset();
        Find.TickManager = new TickManager();
        Rand.Current = new RandomStream(seed);
        Pawn.ResetThingIdCounter();
        SimWorld.Map.Map.ResetMapIdCounter();
        Ablation.Clear();

        DifficultyDef? difficulty = threats ? null : new DifficultyDef
        {
            defName = "MediumWithoutThreats",
            threatScale = DifficultyDefOf.Medium.threatScale,
            allowBigThreats = false,
            allowIntroThreats = false,
        };

        Game game = Game.NewGame(
            ScenarioDefOf.TribalStart.scenario, seed.ToString(CultureInfo.InvariantCulture),
            subdivisionOverride: 3, soloStart: false, bandSize: band, difficultyDef: difficulty);
        SimWorld.World.Settlement home = game.CivilizationTarget.Seat!;
        GodCommands.OpenSettlement(home.tile);

        var infs = new Dictionary<Hediff, Inf>();
        var burnsSeen = new Dictionary<Pawn, int>();
        int lastChron = 0;
        int peakFires = 0;

        for (int day = 1; day <= days; day++)
        {
            for (int i = 0; i < GenDate.TicksPerDay; i++)
            {
                game.TickManager.DoSingleTick();
                int now = Find.TickManager.TicksGame;
                if (now % 30 != 0) continue;
                SimWorld.Map.Map? map = home.InteriorMap;

                if (mode == "infection")
                {
                    foreach (Pawn p in home.Citizens.ToList())
                    {
                        if (p.Dead) continue;
                        foreach (Hediff h in p.health.hediffSet.hediffs)
                        {
                            if (h.def.defName != "WoundInfection" || infs.ContainsKey(h)) continue;
                            infs[h] = new Inf { P = p, H = h, Onset = now, JobAtOnset = p.jobs?.curJob?.def?.defName ?? "none" };
                        }
                    }
                    foreach (Inf inf in infs.Values)
                    {
                        if (inf.End >= 0) continue;
                        Pawn p = inf.P;
                        bool present = !p.Dead && p.health.hediffSet.hediffs.Contains(inf.H);
                        if (!present)
                        {
                            inf.End = now;
                            inf.Outcome = p.Dead ? "DIED" : "cleared";
                            continue;
                        }
                        HediffComp_TendDuration? tend = inf.H.TryGetComp<HediffComp_TendDuration>();
                        float imm = p.health.immunity.GetImmunity(inf.H.def);
                        inf.Samples++;
                        if (tend != null && tend.IsTended) inf.Tended++;
                        if (tend != null && tend.totalTendQuality > inf.LastTotalQ + 0.0001f)
                        {
                            inf.Tends++;
                            inf.SumQ += tend.tendQuality;
                            inf.LastTotalQ = tend.totalTendQuality;
                            if (inf.FirstTended < 0) inf.FirstTended = now;
                            inf.Timeline.Add("  +" + ((now - inf.Onset) / 60000f).ToString("0.00", CultureInfo.InvariantCulture) + "d TEND quality " + F(tend.tendQuality)
                                + " (patient in bed: " + (p.InBed() ? "yes" : "no") + ")");
                        }
                        if (p.InBed()) inf.InBed++;
                        if (p.Asleep) inf.Asleep++;
                        if (p.Downed) inf.Downed++;
                        inf.GainSum += p.ImmunityGainSpeed;
                        if (inf.H.Severity > inf.MaxSev) inf.MaxSev = inf.H.Severity;
                        inf.SevAtEnd = inf.H.Severity;
                        inf.ImmAtEnd = imm;
                        if ((now - inf.Onset) % 6000 == 0)
                        {
                            inf.Timeline.Add("  +" + ((now - inf.Onset) / 60000f).ToString("0.00", CultureInfo.InvariantCulture) + "d sev " + F(inf.H.Severity) + " imm " + F(imm)
                                + " tended " + (tend != null && tend.IsTended ? "yes q" + F(tend.tendQuality) : "no")
                                + " job " + (p.jobs?.curJob?.def?.defName ?? "none") + (p.InBed() ? " IN BED" : "") + (p.Downed ? " DOWNED" : "") + (p.Asleep ? " asleep" : "")
                                + " gainSpeed " + F(p.ImmunityGainSpeed));
                        }
                    }
                }
                else if (map != null)
                {
                    int fires = map.listerThings.AllThings.Count(t => t is Fire);
                    if (fires > peakFires) peakFires = fires;
                    if (fires > 0 && now % 600 == 0)
                    {
                        int burning = map.mapPawns.AllPawnsSpawned.Count(p => p.faction == home.faction && !p.Dead && p.RaceProps.Humanlike && HasAttachedFire(p, map));
                        var jobs = map.mapPawns.AllPawnsSpawned.Where(p => p.faction == home.faction && !p.Dead && p.RaceProps.Humanlike)
                            .GroupBy(p => (p.jobs?.curJob?.def?.defName ?? "none") + (p.Downed ? "(downed)" : ""))
                            .OrderByDescending(g => g.Count()).Select(g => g.Key + " " + g.Count());
                        Console.WriteLine("d" + day + " t" + now + " fires " + fires + " citizens-on-fire " + burning
                            + " raining " + (map.weatherManager.CurWeather?.defName ?? "?") + " | " + string.Join(", ", jobs));
                    }
                    foreach (Pawn p in home.Citizens.ToList())
                    {
                        int burns = p.health.hediffSet.hediffs.Count(h => h.def.defName == "Burn");
                        burnsSeen.TryGetValue(p, out int before);
                        if (burns > before)
                        {
                            burnsSeen[p] = burns;
                            if (burns <= 3 || burns % 10 == 0 || p.Dead)
                            {
                                Console.WriteLine("d" + day + " t" + now + " " + p.Label + " burn #" + burns + " job " + (p.jobs?.curJob?.def?.defName ?? "none")
                                    + (p.Downed ? " DOWNED" : "") + (p.Dead ? " DEAD" : "") + " fire-in-cell " + (p.Spawned && map.thingGrid.ThingsListAt(p.Position).Any(t => t is Fire) ? "yes" : "no")
                                    + " attached " + (HasAttachedFire(p, map) ? "yes" : "no"));
                            }
                        }
                    }
                }

                IReadOnlyList<ChronicleEntry> chron = game.Storyteller.Chronicle;
                for (; lastChron < chron.Count; lastChron++)
                {
                    ChronicleEntry e = chron[lastChron];
                    if (e.deathCause.HasValue) Console.WriteLine("d" + day + " t" + e.tick + " DEATH " + e.deathCause.Value + " " + e.targetLabel);
                    else if (mode != "infection") Console.WriteLine("d" + day + " t" + e.tick + " CHRONICLE " + e.incidentDefName);
                }
            }

            SimWorld.Map.Map? m = home.InteriorMap;
            int beds = m == null ? 0 : m.listerThings.AllThings.Count(t => t.def.defName == "Bed");
            int medsOnMap = m == null ? 0 : m.listerThings.ThingsInGroup(SimWorld.Map.ThingRequestGroup.Item).Where(t => t.def.IsMedicine).Sum(t => t.stackCount);
            int medsInStore = home.Stores.Where(s => s.Key.IsMedicine).Sum(s => s.Value);
            Console.WriteLine("== end of day " + day + ": citizens " + home.Citizens.Count(c => !c.Dead) + ", beds " + beds + ", medicine on map " + medsOnMap
                + ", medicine in ledger " + medsInStore + ", peak fires " + peakFires + ", " + game.Storyteller.deaths);
        }

        if (mode == "infection")
        {
            Console.WriteLine();
            Console.WriteLine("## infections: pawn | onset day | job at onset | first tend after (h) | tends | mean tend quality | share of time tended | in bed | asleep | downed | mean immunity gain speed | max severity | immunity at end | lasted (d) | outcome");
            foreach (Inf inf in infs.Values.OrderBy(x => x.Onset))
            {
                int n = Math.Max(inf.Samples, 1);
                int end = inf.End >= 0 ? inf.End : Find.TickManager.TicksGame;
                Console.WriteLine(inf.P.Label + " | " + (inf.Onset / 60000f).ToString("0.0", CultureInfo.InvariantCulture) + " | " + inf.JobAtOnset + " | "
                    + (inf.FirstTended < 0 ? "never" : ((inf.FirstTended - inf.Onset) / 2500f).ToString("0.0", CultureInfo.InvariantCulture)) + " | " + inf.Tends + " | "
                    + (inf.Tends == 0 ? "-" : F(inf.SumQ / inf.Tends)) + " | " + F(inf.Tended / (float)n) + " | " + F(inf.InBed / (float)n) + " | " + F(inf.Asleep / (float)n) + " | "
                    + F(inf.Downed / (float)n) + " | " + F(inf.GainSum / n) + " | " + F(inf.MaxSev) + " | " + F(inf.ImmAtEnd) + " | "
                    + ((end - inf.Onset) / 60000f).ToString("0.00", CultureInfo.InvariantCulture) + " | " + inf.Outcome);
            }
            foreach (Inf inf in infs.Values.OrderBy(x => x.Onset).Where(x => x.Outcome == "DIED").Take(2))
            {
                Console.WriteLine();
                Console.WriteLine("## fatal case: " + inf.P.Label + ", infection from day " + (inf.Onset / 60000f).ToString("0.00", CultureInfo.InvariantCulture));
                foreach (string line in inf.Timeline) Console.WriteLine(line);
            }
        }
        return 0;
    }

    private static bool HasAttachedFire(Pawn p, SimWorld.Map.Map map) =>
        p.Spawned && map.thingGrid.ThingsListAt(p.Position).Any(t => t is Fire f && ReferenceEquals(f.parent, p));
}
